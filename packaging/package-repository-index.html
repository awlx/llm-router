llm-router Package Repository

Debian and Ubuntu

sudo install -d -m 0755 /etc/apt/keyrings
curl -fsSL https://awlx.github.io/llm-router/llm-router.gpg.key | sudo gpg --dearmor -o /etc/apt/keyrings/llm-router.gpg
echo "deb [signed-by=/etc/apt/keyrings/llm-router.gpg] https://awlx.github.io/llm-router stable main" | sudo tee /etc/apt/sources.list.d/llm-router.list
sudo apt update
sudo apt install llm-router
